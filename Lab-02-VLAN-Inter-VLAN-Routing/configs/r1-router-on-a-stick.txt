! R1 — adapt interface name to your Packet Tracer router model.
! One physical link to SW1 G0/1 carries VLANs 10, 20, 30 (optional), and 99.
enable
configure terminal
hostname R1
no ip domain-lookup

interface gigabitEthernet0/0
 description 802.1Q trunk to SW1 G0/1
 no shutdown
exit

interface gigabitEthernet0/0.10
 encapsulation dot1Q 10
 description Staff gateway
 ip address 192.168.10.1 255.255.255.0
 ip access-group STAFF-IN in
exit

interface gigabitEthernet0/0.20
 encapsulation dot1Q 20
 description Finance gateway
 ip address 192.168.20.1 255.255.255.0
exit

! Optional shared-services VLAN; omit this subinterface and ACL permits if unused.
interface gigabitEthernet0/0.30
 encapsulation dot1Q 30
 description Shared services gateway
 ip address 192.168.30.1 255.255.255.0
exit

interface gigabitEthernet0/0.99
 encapsulation dot1Q 99
 description Management gateway
 ip address 192.168.99.1 255.255.255.0
exit

ip access-list extended STAFF-IN
 remark Permit DNS and web to optional shared server only
 permit udp 192.168.10.0 0.0.0.255 host 192.168.30.10 eq 53
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.30.10 eq 53
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.30.10 eq 80
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.30.10 eq 443
 remark Block Staff initiating connections to Finance
 deny ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
 remark Permit remaining routed traffic for this lab example
 permit ip 192.168.10.0 0.0.0.255 any
exit

end
show ip interface brief
show ip route
show access-lists
! Save after verifying: copy running-config startup-config
